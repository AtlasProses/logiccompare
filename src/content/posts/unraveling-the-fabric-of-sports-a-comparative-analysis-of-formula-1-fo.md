interface FootballPlayer {
    name: string;

        goals: number;

class FootballPlayerAnalyzer {
    private players: FootballPlayer[];

    constructor(players: FootballPlayer[]) {
        this.players = players;
    }

    calculatePlayerRating(player: FootballPlayer): number {
        return (player.stats.goals + player.stats.assists) / player.stats.appearances;
    }

    // Example usage:
    const players: FootballPlayer[] = [
        { name: "Bruno Fernandes", position: "Midfielder", stats: { goals: 10, assists: 15, appearances: 30 } },
    ];
